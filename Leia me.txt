FORMATANDO STRING

Alguns exemplos:

length   -> quantos caracteres a string tem
toUpperCase() -> tudo para Maiusculas
toLowerCase() -> tudo para minusculo

   <script>
        var nome = window.prompt('Qual é seu nome? ');
        document.write(`Olá ${nome}. Seu nome tem ${nome.length} letras.<br>`);
        document.write(`Seu nome em minusculas é ${nome.toLowerCase()}<br>`);
        document.write(`Seu nome em maiusculas é ${nome.toUpperCase()}<br>`);

    </script>

==== ARRAY =====

let num = [5, 8, 2, 9, 3]
//num[5] = 6 //Acrescenta um valor no vetor, especificando a posição
//num.push(9) //Acresenta um valor no vetor na ultima posição
//num.length  //Mostrar a quantidade de elementos que tem no vetor
//num.sort()  //Organiza em ordem crescente
//console.log(num.sort())
//console.log(num[0])
for(let pos=0; pos<num.length; pos++){
    console.log(`A posição ${pos} tem o valor ${num[pos]}`)
}


//Comando para ler array e objetos de forma simplificada
for(let pos in num){
    console.log(`A posição ${pos} tem o valor ${num[pos]}`)
}
